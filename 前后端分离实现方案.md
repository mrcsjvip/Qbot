## 二、技术选型

### 2.1 API框架选择

**推荐：FastAPI**

**优势**:
- ✅ 高性能（基于Starlette和Pydantic）
- ✅ 自动生成API文档（Swagger/OpenAPI）
- ✅ 类型提示支持（Pydantic）
- ✅ 异步支持（async/await）
- ✅ 易于集成现有Python代码

**备选：Flask**
- ⚠️ 性能略低于FastAPI
- ✅ 生态成熟
- ✅ 学习曲线平缓

## 三、API接口设计

### 3.1 回测相关API

#### 3.1.1 创建回测任务

**接口**: `POST /api/v1/backtest`

**请求体**:
```json
{
  "code": "600000.SH",
  "start_date": "2020-01-01",
  "end_date": "2023-12-31",
  "strategy": "bigger_than_ema",
  "initial_cash": 100000,
  "commission": 0.001,
  "benchmark": "000300.SH",
  "params": {
    "ma_period": 20
  }
}
```

**响应**:
```json
{
  "task_id": "bt_20231201_123456",
  "status": "pending",
  "created_at": "2023-12-01T12:34:56"
}
```

#### 3.1.2 查询回测结果

**接口**: `GET /api/v1/backtest/{task_id}`

**响应**:
```json
{
  "task_id": "bt_20231201_123456",
  "status": "completed",
  "results": {
    "total_return": 0.25,
```

### 4.3 服务层（封装现有逻辑）

**文件**: `qbot/api/services/backtest_service.py`

```python
"""
回测服务层 - 封装现有的回测引擎
"""
import uuid
from datetime import datetime
from typing import Dict, Any, Optional

from qbot.common.logging.logger import LOGGER as logger
from qbot.engine.backtest.backtest_main import BacktestEngine  # 复用现有代码
from qbot.api.schemas.backtest import BacktestCreateRequest, BacktestResult


class BacktestService:
    """回测服务"""
    
    def __init__(self):
        self.backtest_engine = BacktestEngine()
        self.tasks: Dict[str, Dict[str, Any]] = {}  # 任务存储（实际应使用数据库）
    
    def create_backtest_task(self, request: BacktestCreateRequest) -> Dict[str, Any]:
        """
        创建回测任务
        
        复用现有的回测引擎逻辑
        """
        task_id = f"bt_{datetime.now().strftime('%Y%m%d_%H%M%S')}_{uuid.uuid4().hex[:6]}"
        
        # 准备回测参数（复用现有代码的参数格式）
        backtest_params = {
            "code": request.code,
            "start_date": request.start_date,
            "end_date": request.end_date,
            "strategy": request.strategy,
            "initial_cash": request.initial_cash,
            "commission": request.commission,
            "benchmark": request.benchmark,
            "params": request.params or {}
        }
        
        # 存储任务信息
        self.tasks[task_id] = {
            "task_id": task_id,
            "status": "pending",
            "params": backtest_params,
            "created_at": datetime.now(),
            "result": None
        }
        
        # 异步执行回测（实际应使用任务队列如Celery）
        self._run_backtest(task_id, backtest_params)
        
        return {
            "task_id": task_id,
            "status": "pending",
            "created_at": self.tasks[task_id]["created_at"]
        }
    
    def _run_backtest(self, task_id: str, params: Dict[str, Any]):
        """
        执行回测（复用现有回测引擎）
        """
        try:
            self.tasks[task_id]["status"] = "running"
            
            # 调用现有的回测引擎（完全复用现有代码）
            result = self.backtest_engine.run(
                code=params["code"],
                start_date=params["start_date"],
                end_date=params["end_date"],
                strategy=params["strategy"],
                initial_cash=params["initial_cash"],
                commission=params["commission"],
                benchmark=params.get("benchmark"),
                strategy_params=params.get("params", {})
            )
            
            # 格式化结果
            formatted_result = self._format_result(result)
            
            self.tasks[task_id]["status"] = "completed"
            self.tasks[task_id]["result"] = formatted_result
            self.tasks[task_id]["completed_at"] = datetime.now()
            
        except Exception as e:
            logger.error(f"回测任务 {task_id} 执行失败: {e}")
            self.tasks[task_id]["status"] = "failed"
            self.tasks[task_id]["error"] = str(e)
    
    def _format_result(self, result: Dict[str, Any]) -> Dict[str, Any]:
        """
        格式化回测结果
        """
        return {
            "total_return": result.get("total_return", 0),
            "annual_return": result.get("annual_return", 0),
            "sharpe_ratio": result.get("sharpe_ratio", 0),
            "max_drawdown": result.get("max_drawdown", 0),
            "win_rate": result.get("win_rate", 0),
            "total_trades": result.get("total_trades", 0),
            "trades": result.get("trades", []),
            "equity_curve": result.get("equity_curve", [])
        }
    
    def get_backtest_result(self, task_id: str) -> Optional[BacktestResult]:
        """
        获取回测结果
        """
        if task_id not in self.tasks:
            return None
        
        task = self.tasks[task_id]
        
        if task["status"] != "completed":
            return {
                "task_id": task_id,
                "status": task["status"],
                "result": None
            }
        
        return BacktestResult(
            task_id=task_id,
            status=task["status"],
            **task["result"],
            created_at=task["created_at"],
            completed_at=task.get("completed_at")
        )
    
    def list_backtests(self, page: int = 1, page_size: int = 20, status: Optional[str] = None):
        """
        获取回测列表
        """
        tasks = list(self.tasks.values())
        
        if status:
            tasks = [t for t in tasks if t["status"] == status]
        
        # 分页
        start = (page - 1) * page_size
        end = start + page_size
        
        return {
            "total": len(tasks),
            "page": page,
            "page_size": page_size,
            "tasks": tasks[start:end]
        }
```

```python
"""

from typing import Dict, Any, Optional, List

    """交易服务"""

        self.trade_engines: Dict[str, TradeEngine] = {}  # 存储交易引擎实例
    
    def start_trade(self, request: TradeStartRequest, user_id: str) -> Dict[str, Any]:
        """
        启动交易（复用现有交易引擎）
        """
        # 创建交易引擎实例（复用现有代码）
        trade_engine = TradeEngine(
            trade_opts=request.trade_opts,
            syslog_obj=None  # 可以传入日志对象
        )
        
        # 登录
        trade_engine.login()
        
        # 启动交易
        trade_engine.start_trade()
        
        # 存储引擎实例（实际应使用更安全的方式）
        engine_id = f"{user_id}_{request.code}"
        self.trade_engines[engine_id] = trade_engine
        
        return {
            "engine_id": engine_id,
            "status": "running"
        }
    
    def get_account_info(self, engine_id: str) -> Optional[AccountInfo]:
        """
        获取账户信息（复用现有方法）

        # 调用现有的获取持仓方法
        positions_data = trade_engine.get_positions()
        
        # 格式化返回
        return [
            Position(
                code=pos.get("code"),
```

```python
from fastapi import APIRouter, HTTPException, Depends
from typing import Optional

from qbot.api.schemas.backtest import BacktestCreateRequest, BacktestResponse, BacktestResult
from qbot.api.services.backtest_service import BacktestService

router = APIRouter()
backtest_service = BacktestService()


@router.post("", response_model=BacktestResponse)
async def create_backtest(request: BacktestCreateRequest):
    """
    创建回测任务
    """
    try:
        result = backtest_service.create_backtest_task(request)
        return result
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@router.get("/{task_id}", response_model=BacktestResult)
async def get_backtest_result(task_id: str):
    """
    获取回测结果
    """
    result = backtest_service.get_backtest_result(task_id)
    if result is None:
        raise HTTPException(status_code=404, detail="回测任务不存在")
    return result


@router.get("")
async def list_backtests(
    page: int = 1,
    page_size: int = 20,
    status: Optional[str] = None
):
    """
    获取回测列表
    """
    return backtest_service.list_backtests(page, page_size, status)
```
